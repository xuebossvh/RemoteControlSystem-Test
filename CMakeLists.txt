cmake_minimum_required(VERSION 3.24)
project(RemoteControlSystem VERSION 1.1.1 LANGUAGES CXX RC)
set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} /c65001")
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC and Win32: cmake -S . -B build -A Win32")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(RCS_ATLMFC_ROOT "${CMAKE_SOURCE_DIR}/.deps/atlmfc" CACHE PATH "MFC/ATL include and lib root (optional for installed MFC)")
if(EXISTS "${RCS_ATLMFC_ROOT}/include/afxwin.h")
    include_directories(SYSTEM "${RCS_ATLMFC_ROOT}/include")
    link_directories("${RCS_ATLMFC_ROOT}/lib/x86")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32 WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _SCL_SECURE_NO_WARNINGS)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W3>" "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" "$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")

set(src "${CMAKE_SOURCE_DIR}/RemoteCtrolSystem")
function(rcs_sources target folder)
    foreach(file IN LISTS ARGN)
        target_sources(${target} PRIVATE "${src}/${folder}/${file}")
    endforeach()
    target_include_directories(${target} PRIVATE "${src}/${folder}")
endfunction()

add_library(Thread STATIC)
rcs_sources(Thread Thread ThreadImpl.cpp AutoLock.cpp GlobalMutex.cpp LocalMutex.cpp)
add_library(Message STATIC)
rcs_sources(Message Message ByteArrayMsg.cpp ImageMsg.cpp KeyboardMsg.cpp MessageData.cpp MouseMsg.cpp SetupMsg.cpp UserMsg.cpp)
add_library(SocketIPv4 STATIC)
rcs_sources(SocketIPv4 SocketIPv4 TCPClient.cpp TCPServer.cpp TCPThreadedServer.cpp trace.cpp TCPSocket.cpp SocketAddress.cpp TCPStringBuffer.cpp TCPConnection.cpp)
target_link_libraries(SocketIPv4 PUBLIC Message Thread ws2_32)
add_library(Graphic STATIC)
rcs_sources(Graphic Graphic GdiImage.cpp ImageTile.cpp ScreenDisplay.cpp ScreenShot.cpp)
target_compile_options(Graphic PRIVATE /utf-8)
target_link_libraries(Graphic PUBLIC Message gdiplus ole32 gdi32 user32)
add_library(tinyxml STATIC)
rcs_sources(tinyxml tinyxml tinystr.cpp tinyxml.cpp tinyxmlerror.cpp tinyxmlparser.cpp)
add_library(ClientConfig STATIC)
rcs_sources(ClientConfig ClientConfig HostListConfig.cpp stdafx.cpp)
target_link_libraries(ClientConfig PUBLIC tinyxml)

add_executable(Client WIN32)
rcs_sources(Client Client AddIpDlg.cpp AddIpSL.cpp CaptureExclusion.cpp Client.cpp ClientDlg.cpp ClientMain.cpp DisplayDlg.cpp RemoteCanvas.cpp FullscreenBar.cpp LoginDialog.cpp ServerListDlg.cpp SetupDlg.cpp stdafx.cpp Client.rc)
target_sources(Client PRIVATE "${src}/Common/LocalDesktopInput.cpp")
target_link_libraries(Client PRIVATE Graphic SocketIPv4 ClientConfig comctl32 shlwapi iphlpapi dwmapi)
add_executable(Server WIN32)
rcs_sources(Server RemoteCtrolSystemServer AddEditIpDlg.cpp AddUserDlg.cpp AES.cpp ChangeAdminPwd.cpp Config.cpp DealConnection.cpp EnterDlg.cpp FileSystemWatcher.cpp IniFile.cpp IpRuleFile.cpp MouseKeybdEvent.cpp OperationDlg.cpp OperationNotes.cpp RemoteCtrolSystemServer.cpp RemoteCtrolSystemServerDlg.cpp SendImageThread.cpp stdafx.cpp SystemMonitor.cpp TabAboutDlg.cpp TabAccountDlg.cpp TabConnectionDlg.cpp TabServerDlg.cpp UserFile.cpp WndClassList.cpp RemoteCtrolSystemServer.rc)
target_link_libraries(Server PRIVATE Graphic SocketIPv4 comctl32 shlwapi crypt32)

foreach(target Client Server)
    target_link_options(${target} PRIVATE /ENTRY:wWinMainCRTStartup)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/${target}-data")
endforeach()
# Copy examples only if missing so rebuilding never overwrites local settings/accounts.
add_custom_command(TARGET Client POST_BUILD COMMAND ${CMAKE_COMMAND}
    "-DSOURCE=${src}/Client/ClientConfig.xml"
    "-DDEST=$<TARGET_FILE_DIR:Client>/Client-data/ClientConfig.xml"
    -P "${CMAKE_SOURCE_DIR}/scripts/copy-if-missing.cmake")
add_custom_command(TARGET Server POST_BUILD COMMAND ${CMAKE_COMMAND}
    "-DSOURCE=${CMAKE_SOURCE_DIR}/config/RemoteConfig.ini"
    "-DDEST=$<TARGET_FILE_DIR:Server>/Server-data/RemoteConfig.ini"
    -P "${CMAKE_SOURCE_DIR}/scripts/copy-if-missing.cmake")
